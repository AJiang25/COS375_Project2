make clean && make sim_cycle tests

run_test () {
  name="$1"
  cfg="$2"
  echo
  echo "=============================="
  echo "RUN ${name}  (config: ${cfg})"
  echo "=============================="
  ./sim_cycle "test/${name}.bin" "${cfg}"
  echo "--- sim_stats ---"
  cat "test/${name}_cycle_sim_stats.out"
  echo "--- reg_state ---"
  cat "test/${name}_cycle_reg_state.out"
}

# run every assembly test in test/ (auto-pick config)
DEFAULT_CFG="test/cache_config.txt"
LRU_CFG="test/cache_config_lru.txt"


for s in test/*.s; do
  name="$(basename "${s}" .s)"
  cfg="${DEFAULT_CFG}"
  if [ "${name}" = "dcache_lru" ]; then
    cfg="${LRU_CFG}"
  fi
  run_test "${name}" "${cfg}"
done
